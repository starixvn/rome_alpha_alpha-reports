Get DCS Locator
    [Arguments]    ${fieldType}=${EMPTY}    ${fieldName}=${EMPTY}    ${product}=${EMPTY}    ${sportType}=${EMPTY}    ${betType}=${EMPTY}
    [Documentation]    This action is used to return Locator of PT Setting Controls:
    ...
    ...    - ${fieldType}: combobox, checkbox, lblA1, lblM1
    ...
    ...    - ${fieldName}: \ filed name need to mention for combobox (pt selected list)
    ...    \ \ \ \ \ \ \ \ \ \ \ \ \ \ \ \ \ \ \ \ \ \ \ \ \ \ \ \ \ a1, m2, uM2, a2, n2, m1, uM1, s3, uS3, m3, n3, s1, uS1
    ...
    ...    - ${product}: product Id
    ...
    ...    - ${sportType}: sport type Id
    ...
    ...    - ${bettype}: bet type Id
    ${a}    Set Variable    ${product}
    ${ac_locator}    set variable    ${in_m_DCS_pt_${fieldType}}
    ${ac_locator}    replace string    ${ac_locator}    uiProduct    ${product}
    ${ac_locator}    replace string    ${ac_locator}    uiSportType    ${sportType}
    ${ac_locator}    replace string    ${ac_locator}    uiBetType    ${betType}
    ${ac_locator_cbo}    run keyword if    '${fieldType}'=='combobox'    replace string    ${ac_locator}    uiFieldName    ${fieldName}
    ${rt_locator}    Set Variable if    '${fieldType}'=='combobox'    ${ac_locator_cbo}    ${ac_locator}
    [Return]    ${rt_locator}
