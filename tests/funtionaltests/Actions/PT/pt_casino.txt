Get Casino PT
    [Arguments]    ${type}
    Select Frame    main
    ${ls_output}    Create List
    # Get Super PT
    ${tm_SuperPresetPT}    Run Keyword If    '${type}'=='Master'    Get Selected List Label    ${in_m_rng_superPrePT}
    ${tm_MasterMaxPT}    Run Keyword If    '${type}'=='Master'    Get Selected List Label    ${in_m_rng_masterMaxPT}
    ${tm_MasterMinPT}    Run Keyword If    '${type}'=='Master'    Get Selected List Label    ${in_m_rng_masterMinPT}
    Run Keyword If    '${type}'=='Master'    Append To List    ${ls_output}    ${tm_SuperPresetPT}    ${tm_MasterMaxPT}    ${tm_MasterMinPT}
    # Get Master PT
    ${tm_MasterPresetPT}    Run Keyword If    '${type}'=='Agent'    Get Selected List Label    ${in_m_rng_masterPrePT}
    ${tm_AgentMaxPT}    Run Keyword If    '${type}'=='Agent'    Get Selected List Label    ${in_m_rng_agentMaxPT}
    ${tm_AgentMinPT}    Run Keyword If    '${type}'=='Agent'    Get Selected List Label    ${in_m_rng_agentMinPT}
    Run Keyword If    '${type}'=='Agent'    Append To List    ${ls_output}    ${tm_MasterPresetPT}    ${tm_AgentMaxPT}    ${tm_AgentMinPT}
    # Get Agent PT
    ${tm_AgentPresetPT}    Run Keyword If    '${type}'=='Member'    Get Selected List Label    ${in_m_rng_agentPrePT}
    Run Keyword If    '${type}'=='Member'    Append To List    ${ls_output}    ${tm_AgentPresetPT}
    Unselect Frame
    [Return]    ${ls_output}

Edit Casino PT
    [Arguments]    ${type}    ${ls_value}
    Unselect Frame
    Select Frame    main
    #SET NORMAL VALUE
    @{ls_value}    Set Variable    ${ls_value}
    Comment    Run Keyword If    '${type}' == 'Master'    Run Keywords    Select Value From List    ${in_m_rng_superPrePT}    @{ls_value}[0]
    ...    AND    Select Value From List    ${in_m_rng_masterMaxPT}    @{ls_value}[1]
    ...    AND    Select Value From List    ${in_m_rng_masterMinPT}    @{ls_value}[2]
    Comment    Run Keyword If    '${type}' == 'Agent'    Run Keyword    Edit Casino PT Agent    ${in_m_CS_pt_cboM2}    @{ls_value}[0]
    ...    AND    Select Value From List    ${in_m_rng_agentMaxPT}    @{ls_value}[1]
    ...    AND    Select Value From List    ${in_m_rng_agentMinPT}    @{ls_value}[2]
    Comment    Run Keyword If    '${type}' == 'Agent'    Run Keyword    Select Value From List    ${in_m_CS_pt_cboA2}    @{ls_value}[1]
    Comment    Run Keyword If    '${type}' == 'Member'    Run Keyword    Select Value From List    ${in_m_CS_pt_cboA1}    @{ls_value}[0]
    Comment    Sleep    2s
    Comment    Click Element    ${in_m_CS_btnSubmit}
    Comment    Sleep    10s
    Run Keyword If    '${type}' == 'MasterAgent'    Edit Casino PT_MasterAgent
    Run Keyword If    '${type}' == 'AgentMember'    Edit Casino PT_AgentPlayer    ${ls_value}
    Unselect Frame

Check Casino PT
    [Arguments]    ${type}    @{expect}
    select link in setting tab    Edit Casino
    ${tm_Super}    Run Keyword If    '${type}' == 'Master'    Get Casino PT    Master
    ${tm_Master}    Run Keyword If    '${type}' == 'Agent'    Get Casino PT    Agent
    ${tm_Agent}    Run Keyword If    '${type}' == 'Member'    Get Casino PT    Member
    Run Keyword If    '${type}' == 'Master'    Lists Should Be Equal    ${tm_Super}    @{expect}
    Run Keyword If    '${type}' == 'Agent'    Lists Should Be Equal    ${tm_Master}    @{expect}
    Run Keyword If    '${type}' == 'Member'    Lists Should Be Equal    ${tm_Agent}    @{expect}

Edit RNG Casino PT
    [Arguments]    ${action}=${EMPTY}    ${ptName}=${EMPTY}    ${value}=${EMPTY}
    [Documentation]    This action is used to update Casino PT

    Select Frame    main
