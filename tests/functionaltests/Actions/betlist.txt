BetList_Check Common Values
    [Arguments]    ${refno}    ${OddsValues}    ${Stake}    ${WinlossValues}    ${StatusValues}    ${AgentPTComm}
    ...    ${MasterPTComm}=${EMPTY}    ${SuperPTComm}=${EMPTY}
    [Documentation]    This action is used to check bet list value excluding choice values
    ...
    ...    ${refno}: the refno/transid of ticket
    ...    ${OddsValues}: the odds value and oddstype separated by ";"
    ...    ${Stake}: the stake value
    ...    ${WinlossValues}: the winloss and comm values separated by ";"
    ...    ${Status}: the status,result link and IP values separated by ";"
    ...    ${AgentPTComm}: the PT%, \ PT value \ and Comm% and comm value of Agent, \ separated by ";"
    ...    ${MasterPTComm}: the PT%, \ PT value \ and Comm% and comm value of Master, separated by ";" \ (${EMPTYP} mean no need to check)
    ...    ${SuperPTComm}: the PT%, \ PT value \ and Comm% and comm value of Super, \ separated by ";" \ (${EMPTYP} mean no need to check)

    ${acRow}    Set Variable    //tr[@id='r${refno}']//
    ${aclblOdds}    Replace String    ${lblOdds}    xpath=//    ${acRow}
    ${aclblOddsType}    Replace String    ${lblOddsType}    xpath=//    ${acRow}
    ${aclblStake}    Replace String    ${lblStake}    xpath=//    ${acRow}
    ${aclblWinloss}    Replace String    ${lblWinloss}    xpath=//    ${acRow}
    ${aclblComm}    Replace String    ${lblComm}    xpath=//    ${acRow}
    ${aclblStatus}    Replace String    ${lblStatus}    xpath=//    ${acRow}
    ${aclnkResult}    Replace String    ${lnkResult}    xpath=//    ${acRow}
    ${aclnkIP}    Replace String    ${lnkIP}    xpath=//    ${acRow}
    ## Get checked value
    @{actmpValues}    Split String    ${OddsValues}    ;
    ##check odds column
    Run Keyword And Continue On Failure    Element Text Should Be    ${aclblOdds}    @{actmpValues}[0]
    Run Keyword If    len(@{actmpValues})>1    Run Keyword And Continue On Failure    Element Text Should Be    ${aclblOddsType}    @{actmpValues}[1]
    ##check Stake
    Run Keyword And Continue On Failure    Element Text Should Be    ${aclblStake}    ${Stake}
    ##check Winloss
    @{actmpValues}    Split String    ${WinlossValues}    ;
    Run Keyword And Continue On Failure    Element Text Should Be    ${aclblWinloss}    @{actmpValues}[0]
    Run Keyword And Continue On Failure    Element Text Should Be    ${aclblComm}    @{actmpValues}[1]
    ##check Status, Result, IP
    @{actmpValues}    Split String    ${StatusValues}    ;
    Run Keyword And Continue On Failure    Element Text Should Be    ${aclblStatus}    @{actmpValues}[0]
    Run Keyword And Continue On Failure    Element Text Should Be    ${aclnkResult}    @{actmpValues}[1]
    Run Keyword And Continue On Failure    Element Text Should Be    ${aclnkIP}    @{actmpValues}[2]

    Run Keyword If    '${SuperPTComm}'<>'${EMPTY}'    BetList_CheckSMAPTComm    ${refno}    Super    ${SuperPTComm}
    ##Master
    Run Keyword If    '${MasterPTComm}'<>'${EMPTY}'    BetList_CheckSMAPTComm    ${refno}    Master    ${MasterPTComm}
    ## Agent
    Run Keyword If    '${AgentPTComm}'<>'${EMPTY}'    BetList_CheckSMAPTComm    ${refno}    Agent    ${AgentPTComm}
