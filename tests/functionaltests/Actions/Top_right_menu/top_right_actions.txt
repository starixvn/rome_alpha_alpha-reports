Get Log ID
    [Arguments]    ${date}    ${action}    ${variance}
    @{logID}=    Execute Javascript    var logID,idate,iaction,date,action, variance; \ date = ${date}; \ action = "${action}"; \ variance = ${variance}; logID = new Array(); \ tbl = document.getElementById("DataLog"); \ if (tbl == null || tbl == undefined) return logID; \ noR= tbl.rows.length; \ for(i=0;i<noR;i++) \ \ { \ \ \ cells = tbl.rows[i].cells; \ \ \ idate = getText(cells[1]).trim(); \ \ \ idate = new Date(idate); \ \ \ idate = (idate.getTime())/1000; \ \ \ gap = Math.abs(date - idate); \ \ \ iaction = getText(cells[3]).trim(); \ \ \ if(gap <= variance && (iaction.localeCompare(action)==0)) \ \ \ { \ \ \ \ logID.push(getText(cells[0])); \ \ \ } \ \ } \ return logID; function getText(e){ \ return (e.innerText != undefined) ? e.innerText : e.textContent; \ }
    [Return]    @{logID}
