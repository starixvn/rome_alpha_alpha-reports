Get Racing PT controls
    [Arguments]    ${type}    ${Race}    ${Tote}    ${bettype}
    [Documentation]    This action is used to get Racing PT Controls
    ...
    ...    ${acc_output}: base ${type}
    ...    - ${type}= SuperMaster: S3, Is3, Us3, M3
    ...    - ${type}= SuperPlayer: S1, Is1, Us1
    ...    - ${type}= MasterAgent: M2, Im2, Um2, A2
    ...    - ${type}= MasterPlayer: M1, Im1, Um1
    ...    - ${type}= AgentPlayer: A1
    ${ac_sportID}=    Get From Dictionary    ${SporttypeDict}    ${Race}
    ${ac_DictBettype}    Set Variable If    '${bettype}'=='Win'    ${ToteDictWin}    ${ToteDictPlace}
    ${ac_BettypeID}    Get From Dictionary    ${ac_DictBettype}    ${Tote}
    ${acc_suffix}    Set Variable    $${ac_sportID}$${ac_bettypeID}
    ${acc_lstSuperMaster}    Create List    s3${acc_suffix}    is3${acc_suffix}    us3${acc_suffix}    m3${acc_suffix}
    ${acc_lstSuperPlayer}    Create List    s1${acc_suffix}    is1${acc_suffix}    us1${acc_suffix}
    ${acc_lstMasterAgent}    Create List    m2${acc_suffix}    im2${acc_suffix}    um2${acc_suffix}    a2${acc_suffix}
    ${acc_lstMasterPlayer}    Create List    m1${acc_suffix}    im1${acc_suffix}    um1${acc_suffix}
    ${acc_lstAgentPlayer}    Create List    a1${acc_suffix}
    ${acc_output}    Get Variable Value    ${acc_lst${type}}
    [Return]    ${acc_output}

Edit Racing PT
    [Arguments]    ${username}    ${type}    ${Race}    ${Tote}    ${bettype}    ${lstValues}
    ...    ${isUpdated}=1
    [Documentation]    This action is used to edit Sportsbook PT of account
    ...
    ...    ${username}: the username of account
    ...
    ...    ${type}: the account type (Master/Agent/Member)
    ...
    ...    ${Race}: the race name
    ...
    ...    ${Tote}: the Tote name
    ...
    ...    ${Bettype}: the bettype
    ...
    ...    ${lstValues}: the list contain PT value (MaxPT, UplinePresetPT,isUplineAutoPT,UplineAutoPT). The ${Empty} item mean this item does not need to be edited
    ...
    ...    ${isUpdated}: click update button or not (0/1)
    ...
    ...    ${acc_output}: the output msg
    ## Goto Edit page
    Navigate to Edit page    Edit Racing    ${username}
    ## Get SB PT Controls
    @{tmplst}    Get Racing PT controls    ${type}    ${Race}    ${Tote}    ${bettype}
    ${NoAddValue}    Evaluate    4 - len(${lstValues})
    Repeat Keyword    ${NoAddValue}    Append To List    ${lstValues}    ${EMPTY}
    ## Get SB PT Values
    @{tmpValues}    Set Variable    ${lstValues}
    ## Set Auto or not
    ${kw}    Set Variable if    '@{tmpValues}[2]' == '1'    Select Checkbox    Unselect Checkbox
    Run Keyword Unless    '@{tmpValues}[2]' == '${EMPTY}'    ${kw}    @{tmplst}[2]
    ## Select PT
    Edit Control value    @{tmplst}[3]    Select From List By Value    @{tmpValues}[3]
    Edit Control value    @{tmplst}[1]    Select From List By Value    @{tmpValues}[1]
    Edit Control value    @{tmplst}[0]    Select From List By Value    @{tmpValues}[0]
    ## Click Update
    Run Keyword If    '${isUpdated}'=='1'    Click Element    ${in_m_editsb_btnUpdate}
    Run Keyword And Continue On Failure    Wait Until Element Is Visible    ${in_m_editsb_msg}    ${cf_timeout_long}
    ## Get msg
    ${acc_output}    Run Keyword And Continue On Failure    Get Text    ${in_m_editsb_msg}
    [Return]    ${acc_output}

Get Racing PT Values
    [Arguments]    ${type}    ${username}    ${Race}    ${Tote}    ${bettype}
    # get sporttype, tote type and PT type
    @{ac_lstControls}    Get Racing PT controls    ${type}    ${Race}    ${Tote}    ${bettype}
    ##
    Navigate to Edit page    Edit Racing    ${username}
    ${acc_output}    Create List
    ${acc_posChk}    Set Variable    2
    ${acc_i}    Set Variable    -1
    : FOR    ${acc_iControl}    IN    @{ac_lstControls}
    \    ${acc_i}    Evaluate    ${acc_i}+1
    \    Continue For Loop If    ${acc_i}==${acc_posChk}
    \    ${acc_iValue}    Execute Javascript    return document.getElementById("${acc_iControl}").value
    \    Append To List    ${acc_output}    ${acc_iValue}
    ${acc_iControl}    Set Variable If    '${type}'=='Member'    ${EMPTY}    @{ac_lstControls}[2]
    ${acc_iValue}    Run Keyword Unless    '${type}'=='Member'    Get Element Attribute    ${acc_iControl}@checked
    ${acc_iValue}    Set Variable If    '${acc_iValue}'=='true'    1    0
    Run Keyword Unless    '${type}'=='Member'    Insert Into List    ${acc_output}    ${acc_posChk}    ${acc_iValue}
    [Return]    ${acc_output}

Check Racing PT
    [Arguments]    ${username}    ${type}    ${Race}    ${Tote}    ${bettype}    ${lstValues}
    [Documentation]    This action is used to check Sportsbook PT of account
    ...
    ...    ${username}: the username of account
    ...
    ...    ${type}: the account type (Master/Agent/Member)
    ...
    ...    ${Race}: the race name
    ...
    ...    ${Tote}: the Tote name
    ...
    ...    ${Bettype}: the bettype
    ...
    ...    ${lstValues}: the list contain PT value (MaxPT, UplinePresetPT,isUplineAutoPT,UplineAutoPT). The ${Empty} item mean this item does not need to be edited
    Navigate to Edit page    Edit Racing    ${username}
    @{tmplst}    Get Racing PT controls    ${type}    ${Race}    ${Tote}    ${bettype}
    ${NoAddValue}    Evaluate    4 - len(${lstValues})
    Repeat Keyword    ${NoAddValue}    Append To List    ${lstValues}    ${EMPTY}
    @{tmpValues}    Set Variable    ${lstValues}
    ${i}    Set Variable    -1
    : FOR    ${ivalue}    IN    @{tmpValues}
    \    ${i}    Evaluate    ${i} + 1
    \    Continue For Loop If    '${ivalue}'=='${EMPTY}'
    \    ${tmpKw}    Set Variable If    ${i}==2 and '${ivalue}'=='0'    Checkbox Should Not Be Selected    Checkbox Should Be Selected
    \    Run Keyword And Continue On Failure    Run Keyword If    ${i}==2    ${tmpKw}    @{tmplst}[${i}]
    \    Continue For Loop If    ${i}==2
    \    Run Keyword And Continue On Failure    List Selection Should Be    @{tmplst}[${i}]    ${ivalue}
