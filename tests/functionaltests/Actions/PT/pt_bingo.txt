Get Bingo PT Values

    Navigate to Edit page    Edit Bingo    ${username}
    @{acc_tmplst}    Get Bingo PT Controls    ${type}
    ${acc_output}    Create List
    ${acc_posChk}    Set Variable    2
    ${acc_i}    Set Variable    -1
    : FOR    ${acc_iControl}    IN    @{acc_tmplst}
    \    ${acc_i}    Evaluate    ${acc_i}+1
    \    Continue For Loop If    ${acc_i}==${acc_posChk}
    \    ${acc_iValue}    Run Keyword Unless    '${acc_iControl}'=='${Empty}'    Execute Javascript    return document.getElementById("${acc_iControl}").value
    \    Append To List    ${acc_output}    ${acc_iValue}
    ${acc_iControl}    Set Variable If    '${type}'=='Master'    @{acc_tmplst}[2]    ${EMPTY}
    ${acc_iValue}    Run Keyword If    '${type}'=='Master'    Get Element Attribute    ${acc_iControl}@checked
    ${acc_iValue}    Set Variable If    '${acc_iValue}'=='true'    1    0
    Run Keyword If    '${type}'=='Master'    Insert Into List    ${acc_output}    ${acc_posChk}    ${acc_iValue}
    [Return]    ${acc_output}

Check Bingo PT
    [Arguments]    ${username}    ${type}    ${lstValues}
    [Documentation]    This action is used to edit Bingo PT of account
    ...
    ...    ${username}: the username of account
    ...
    ...    ${type}: the account type (Master/Agent/Member)
    ...
    ...    ${lstValues}: the list contain PT value (MaxPT, UplinePresetPT,isUplineAutoPT,UplineAutoPT,MinPT). The ${Empty} item mean this item does not need to be edited
    ${acc_currentValues}    Get Bingo PT Values    ${username}    ${type}
    ${acc_lstItemName}    create list    Max PT    Upline Preset PT    isAuto    Upline Auto Preset PT    Min PT
    ${acc_len}    Get Length    ${lstValues}
    ${acc_len}    Evaluate    min(len(${acc_currentValues}),${acc_len})
    : FOR    ${i}    IN RANGE    0    ${acc_len}
    \    ${ichecked}    Get From List    ${lstValues}    ${i}
    \    Continue For Loop If    '${ichecked}'=='' or '${ichecked}'=='${EMPTY}'
    \    ${PT}    Get From List    ${acc_currentValues}    ${i}
    \    ${Item}    Get From List    ${acc_lstItemName}    ${i}
    \    Run Keyword And Continue On Failure    Should Be Equal As Numbers    ${PT}    ${ichecked}    ${Item} value is not correct. Current value is ${PT} but checked value is ${ichecked}

    Navigate to Edit page    Edit Bingo    ${username}
    ## Get SB PT Controls
    @{tmplst}    Get Bingo PT Controls    ${type}
    ${NoAddValue}    Evaluate    5 - len(${lstValues})
    Repeat Keyword    ${NoAddValue}    Append To List    ${lstValues}    ${EMPTY}
    ## Get SB PT Values
    @{tmpValues}    Set Variable    ${lstValues}
    ## Set Auto or not
    ${kw}    Set Variable if    '@{tmpValues}[2]' == '1'    Select Checkbox    Unselect Checkbox
    Run Keyword Unless    '@{tmpValues}[2]' == '${EMPTY}'    ${kw}    @{tmplst}[2]
    ${kw}    Set Variable    Select From List By Value
    ##
    Edit Control value    @{tmplst}[3]    ${kw}    @{tmpValues}[3]
    Edit Control value    @{tmplst}[1]    ${kw}    @{tmpValues}[1]
    Edit Control value    @{tmplst}[0]    ${kw}    @{tmpValues}[0]
    Edit Control value    @{tmplst}[0]    ${kw}    @{tmpValues}[4]
    ## Click Update
    Run Keyword If    '${isUpdated}'=='1'    Click Element    ${in_m_editbingo_btnSubmit}
    Run Keyword And Continue On Failure    Wait Until Element Is Visible    ${in_m_editsb_msg}    ${cf_timeout_long}
    ## Get msg
    Comment    ${acc_output}    Run Keyword And Continue On Failure    Get Text    ${in_m_editsb_msg}
    ${acc_output}    Set Variable    ${EMPTY}
    [Return]    ${acc_output}
