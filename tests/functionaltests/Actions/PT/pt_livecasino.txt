Check Live Casino PT
    [Arguments]    ${type}    ${autopt}    ${submit}    @{lstValues}
    [Documentation]    This action is used to check Live Casino PT of account (include auto PT)
    ...
    ...    ${type}: the edit account type (Master/Agent/Member)
    ...    *** for auto PT type= (A-Master/A-Agent)
    ...
    ...    ${autopt}: value of auto PT (US3/UM2)
    ...
    ...
    ...    ${lstValues}: the list contain PT values:
    ...
    ...    +Master: S3,M3,N3
    ...
    ...    +Agent: M2,A2, N2
    ...
    ...    +Member: A1
    ...
    ...    ${submit}: Yes/no (submit or not)
    Comment    Search Account    ${tm_downline}
    Comment    select link in setting tab    Edit Live Casino
    Select Frame    main
    # Get Master PT
    ${ac_s3}=    Run Keyword If    '${type}'=='Master'    Get Selected List Label    id=ddlS3
    ${ac_m3}=    Run Keyword If    '${type}'=='Master'    Get Selected List Label    id=ddlM3
    ${ac_n3}=    Run Keyword If    '${type}'=='Master'    Get Selected List Label    id=ddlN3
    # Get Agent PT
    ${ac_m2}=    Run Keyword If    '${type}'=='Agent'    Get Selected List Label    id=ddlM2
    ${ac_a2}=    Run Keyword If    '${type}'=='Agent'    Get Selected List Label    id=ddlA2
    ${ac_n2}=    Run Keyword If    '${type}'=='Agent'    Get Selected List Label    id=ddlN2
    # Get Member PT
    ${ac_a1}=    Run Keyword If    '${type}'=='Member'    Get Selected List Label    id=ddlA1
    #Get AutoPT values
    ${ac_us3}    Run Keyword If    '${type}'=='A-Master'    Get Selected List Label    id=ddlUs3
    ${ac_um2}    Run Keyword If    '${type}'=='A-Agent'    Get Selected List Label    id=ddlUm2
    #Check master PT
    Run Keyword If    '${type}'=='Master'    Run Keywords    PT Should Be Equal    ${ac_s3}    @{lstValues}[0]
    ...    AND    PT Should Be Equal    ${ac_m3}    @{lstValues}[1]
    ...    AND    PT Should Be Equal    ${ac_n3}    @{lstValues}[2]
    #Check agent PT
    Run Keyword If    '${type}'=='Agent'    Run Keywords    PT Should Be Equal    ${ac_m2}    @{lstValues}[0]
    ...    AND    PT Should Be Equal    ${ac_a2}    @{lstValues}[1]
    ...    AND    PT Should Be Equal    ${ac_n2}    @{lstValues}[2]
    #Check member PT
    Run Keyword If    '${type}'=='Member'    PT Should Be Equal    ${ac_a1}    @{lstValues}[0]
    #Check auto PT
    Run Keyword If    '${type}'=='A-Master'    Run Keywords    Checkbox Should Be Selected    id=cbIs3
    ...    AND    PT Should Be Equal    ${ac_us3}    ${autopt}
    Run Keyword If    '${type}'=='A-Agent'    Run Keywords    Checkbox Should Be Selected    id=cbIm2
    ...    AND    PT Should Be Equal    ${ac_um2}    ${autopt}
    #click submit
    Run Keyword If    '${submit}'=='Yes'    Click Element    xpath=//*[@value='Update']
    Unselect Frame

Edit Live Casino PT
    [Arguments]    ${type}    ${autopt}    ${submit}    @{lstValues}
    [Documentation]    This action is used to edit Live Casino PT of account
    ...
    ...    ${type}: the edit account type (Master/Agent/Member)
    ...    *** for auto PT type= (A-Master/A-Agent/A-Member)
    ...
    ...    ${autopt}: value of auto PT (US3/UM2/UM1)
    ...
    ...    ${lstValues}: the list contain PT values:
    ...
    ...    +Master: S3,M3,N3
    ...
    ...    +Agent: M2,A2, N2
    ...
    ...    +Member: M1,A1,S1
    ...
    ...    ${submit}: Yes/no (submit or not)
    Unselect Frame
    Select Frame    main
    # set auto PT value
    Run Keyword If    '${type}'=='A-Master'    Run Keywords    Select Checkbox    xpath=//*[contains(@id,'cbI')]
    ...    AND    PT Select From List By Label    xpath=//*[contains(@id,'ddlU')]    ${autopt}
    Run Keyword If    '${type}'=='A-Agent'    Run Keywords    Select Checkbox    xpath=//*[contains(@id,'cbI')]
    ...    AND    PT Select From List By Label    xpath=//*[contains(@id,'ddlU')]    ${autopt}
    Run Keyword If    '${type}'=='A-Member'    Run Keywords    Select Checkbox    xpath=//*[contains(@id,'cbI')]
    ...    AND    PT Select From List By Label    xpath=//*[contains(@id,'ddlU')]    ${autopt}
    #uncheck auto PT    if input type not A-level therefor the action uncheck auto PT will be executed to enable S3/M3/S1...
    ${ac_uncheckpt}    Run Keyword And Return Status    Should Contain    ${type}    A-
    #case edit a1 \ (no auto pt checkbox)
    ${ac_autoptcb}    Run Keyword And Return Status    Element Should Be Visible    xpath=//*[contains(@id,'cbI')]
    ${ac_cbstatus}    Run Keyword And Return Status    Checkbox Should Be Selected    xpath=//*[contains(@id,'cbI')]
    Run Keyword If    '${ac_uncheckpt}'=='False' and '${ac_autoptcb}'=='True' and '${ac_cbstatus}'=='True'    Run Keywords    Unselect Frame
    ...    AND    Uncheck AutoPT
    ...    AND    Select Frame    main
    Comment    Run Keyword If    '${ac_uncheckpt}'=='False'    Unselect Checkbox    xpath=//*[contains(@id,'cbI')]
    #SET NORMAL VALUE
    Run Keyword If    '${type}'=='Master' or '${type}'=='A-Master'    Run Keywords    PT Select From List By Label    id=ddlS3    @{lstValues}[0]
    ...    AND    PT Select From List By Label    id=ddlM3    @{lstValues}[1]
    ...    AND    PT Select From List By Label    id=ddlN3    @{lstValues}[2]
    Run Keyword If    '${type}'=='Agent' or '${type}'=='A-Agent'    Run Keywords    PT Select From List By Label    id=ddlM2    @{lstValues}[0]
    ...    AND    PT Select From List By Label    id=ddlA2    @{lstValues}[1]
    ...    AND    PT Select From List By Label    id=ddlN2    @{lstValues}[2]
    Run Keyword If    '${type}'=='Member' or '${type}'=='A-Member'    Run Keywords    PT Select From List By Label    id=ddlM1    @{lstValues}[0]
    ...    AND    PT Select From List By Label    id=ddlA1    @{lstValues}[1]
    ...    AND    PT Select From List By Label    id=ddlS1    @{lstValues}[2]
    Run Keyword If    '${submit}'=='Yes'    Run Keywords    Click Element    xpath=//*[@value='Update']
    ...    AND    sleep    1s
    ...    AND    Wait Until Page Contains Element    xpath=//*[@class='update-setting-agency-submit']    10s
    Run Keyword If    '${action}'=='save'    Wait Until Page Not Contains Element    ${in_main_iconLoading}
    Unselect Frame
    ${ac_time}    get system time
    [Return]    ${ac_time}

    : FOR    ${i}    IN RANGE    ${ac_tmplength}
    \    ${controlname}    ${ac_new}=    Split String    @{lst1}[${i}]    _
    \    ${controlname}    ${ac_old}=    Split String    @{lst2}[${i}]    _
