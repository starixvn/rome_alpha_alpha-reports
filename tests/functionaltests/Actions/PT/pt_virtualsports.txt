    [Arguments]    ${type}    ${sport}
    [Documentation]    This action is used to get Racing PT Controls
    ...
    ...    ${acc_output}: base ${type}
    ...    - ${type}= SuperMaster: S3, Is3, Us3, M3
    ...    - ${type}= SuperPlayer: S1, Is1, Us1
    ...    - ${type}= MasterAgent: M2, Im2, Um2, A2
    ...    - ${type}= MasterPlayer: M1, Im1, Um1
    ...    - ${type}= AgentPlayer: A1
    ${ac_sportID}=    Get From Dictionary    ${SporttypeDict}    ${sport}
    ${ac_BettypeID}    Get From Dictionary    ${BettypeDict}    ${sport}
    ${acc_suffix}    Set Variable    $${ac_sportID}$${ac_bettypeID}
    ${acc_lstSuperMaster}    Create List    s3${acc_suffix}    is3${acc_suffix}    m3${acc_suffix}
    ${acc_lstSuperPlayer}    Create List    s1${acc_suffix}    is1${acc_suffix}    us1${acc_suffix}
    ${acc_lstMasterAgent}    Create List    m2${acc_suffix}    im2${acc_suffix}    a2${acc_suffix}
    ${acc_lstMasterPlayer}    Create List    m1${acc_suffix}    im1${acc_suffix}
    ${acc_lstAgentPlayer}    Create List    a1${acc_suffix}
    ${acc_output}    Get Variable Value    ${acc_lst${type}}
    [Return]    ${acc_output}
