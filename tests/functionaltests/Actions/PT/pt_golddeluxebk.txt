update PT Super Edit Master
    [Arguments]    ${isUpdate}    ${s3}=${EMPTY}    ${m3}=${EMPTY}    ${is3}=${EMPTY}    ${us3}=${EMPTY}    ${n3}=${EMPTY}
    ...    ${message}=${EMPTY}
    [Documentation]    This allow update master PT: s3,m3,is3,us3,n3
    ...
    ...    ${isUpdate}:
    ...
    ...    '0': NOT Save
    ...
    ...    '1': save all successfully by click Update button
    ...
    ...    '2': save PT/comm group by click icon save
    ...
    ...    '-1': save all UN-successfully
    ...
    ...    '-2': save group UN-succesfully
    ...
    ...    ${Messge}: the message when save success or error. If message is not ${EMPTY} this function will verify message
    Select Frame    main
    Wait Until Element Is Visible    ${in_m_editSetting_cboS3}
    Run Keyword If    '${s3}'!='${EMPTY}'    Select From List By Label    ${in_m_editSetting_cboS3}    ${s3}
    Run Keyword If    '${m3}'!='${EMPTY}'    Select From List By Label    ${in_m_editSetting_cboM3}    ${m3}
    # make sure annoucement is closed before set checkbox
    Unselect Frame
    ${ac_isAnnounAppear}    Run Keyword And Return Status    Page Should Contain Element    ${in_a_btnClosed}
    Run Keyword If    '${ac_isAnnounAppear}'=='True'    Click Element    ${in_a_btnClosed}
    Select Frame    main
    Run Keyword If    '${is3}'!='${EMPTY}'    Set Checkbox    ${in_m_editSetting_chkIs3}    ${is3}
    Run Keyword If    '${us3}'!='${EMPTY}'    Select From List By Label    ${in_m_editSetting_cboUs3}    ${us3}
    Run Keyword If    '${n3}'!='${EMPTY}'    Select From List By Label    ${in_m_editSetting_cboN3}    ${n3}
    Run Keyword If    '${isUpdate}'== '1'    run keywords    Click Element    ${in_m_editSetting_btnUpdate}
    ...    AND    Wait Until Page Contains Element    ${in_m_editSetting_message}
    ...    AND    Wait Until Page Not Contains Element    ${in_m_editSetting_message}
    Run Keyword If    '${isUpdate}'== '2'    run keywords    Click Element    ${in_m_editSetting_btnPTCommSave}
    ...    AND    Wait Until Page Contains Element    ${in_m_editSetting_message}
    ...    AND    Wait Until Page Not Contains Element    ${in_m_editSetting_message}
    Run Keyword If    '${isUpdate}'== '-1'    run keywords    Click Element    ${in_m_editSetting_btnUpdate}
    ...    AND    Wait Until Page Contains Element    ${in_m_editSetting_message}
    Run Keyword If    '${isUpdate}'== '-2'    run keywords    Click Element    ${in_m_editSetting_btnPTCommSave}
    ...    AND    Wait Until Page Contains Element    ${in_m_editSetting_message}
    Unselect Frame
